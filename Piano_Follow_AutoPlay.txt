-- ★ Piano Follow + Auto Play ★
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Workspace = game:GetService("Workspace")
local HttpService = game:GetService("HttpService")
local SG = game:GetService("StarterGui")
local LP = Players.LocalPlayer

local function N(t, x, d)
    pcall(function() SG:SetCore("SendNotification", {Title=t, Text=x, Duration=d or 3}) end)
end

-- ==============================
-- GrabEvents
-- ==============================
local GE = ReplicatedStorage:FindFirstChild("GrabEvents")
local SNO = GE and GE:FindFirstChild("SetNetworkOwner")

-- ==============================
-- MusicKeyboard を探す
-- ==============================
local function GetKeyboard()
    local spawnedToys = Workspace:FindFirstChild(LP.Name .. "SpawnedInToys")
    if spawnedToys then
        local kb = spawnedToys:FindFirstChild("MusicKeyboard")
            or spawnedToys:FindFirstChild("InstrumentPianoKeyboard")
            or spawnedToys:FindFirstChild("InstrumentPianoMelodica")
        if kb then return kb end
    end
    for _, item in ipairs(Workspace:GetChildren()) do
        if item.Name == "MusicKeyboard" or item.Name == "InstrumentPianoKeyboard" or item.Name == "InstrumentPianoMelodica" then
            return item
        end
    end
    return nil
end

-- ==============================
-- 鍵盤を鳴らす
-- ==============================
local function PressKey(kb, keyName)
    if not kb or not SNO then return end
    local key = kb:FindFirstChild(keyName, true)
    if key and key:IsA("BasePart") then
        pcall(function() SNO:FireServer(key, key.CFrame) end)
    end
end

-- ==============================
-- 固定機能
-- ==============================
local FollowActive = false
local FollowConn = nil
local OriginalCollisions = {}

local function StopFollow()
    FollowActive = false
    if FollowConn then FollowConn:Disconnect(); FollowConn = nil end
    for part, cc in pairs(OriginalCollisions) do
        if part and part.Parent then
            pcall(function() part.CanCollide = cc end)
        end
    end
    OriginalCollisions = {}
end

local function StartFollow()
    if FollowActive then return end
    local kb = GetKeyboard()
    if not kb then N("Error", "MusicKeyboardなし", 3) return end
    FollowActive = true

    for _, part in ipairs(kb:GetDescendants()) do
        if part:IsA("BasePart") then
            if OriginalCollisions[part] == nil then
                OriginalCollisions[part] = part.CanCollide
            end
            pcall(function()
                part.CanCollide = false
                part.CanTouch = false
                part.CanQuery = false
                part.Massless = true
                part.Anchored = false
            end)
            pcall(function() part:SetNetworkOwner(LP) end)
        end
    end

    FollowConn = RunService.RenderStepped:Connect(function()
        if not FollowActive then return end
        local kb2 = GetKeyboard()
        if not kb2 then StopFollow(); return end
        local char = LP.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not root then return end
        local mainPart = kb2:FindFirstChild("Main", true) or kb2.PrimaryPart
        if not mainPart then
            for _, part in ipairs(kb2:GetDescendants()) do
                if part:IsA("BasePart") then mainPart = part; break end
            end
        end
        if not mainPart then return end
        local targetCF = root.CFrame * CFrame.new(0, -1.5, -2) * CFrame.Angles(0, math.rad(180), 0)
        pcall(function()
            kb2:PivotTo(targetCF)
            mainPart.AssemblyLinearVelocity = Vector3.zero
            mainPart.AssemblyAngularVelocity = Vector3.zero
        end)
    end)
    N("Follow", "追従開始", 2)
end

    if playing then N("Piano", "演奏中", 2) return end
    local kb = GetKeyboard()
    if not kb then N("Error", "MusicKeyboardなし", 3) return end
    local data = Songs[currentSong]
    if not data then N("Error", "曲データなし", 3) return end
    playing = true
    N("Piano", currentSong .. " 開始", 3)
    task.spawn(function()
        for _, note in ipairs(data) do
            if not playing then break end
            PressKey(kb, note.key)
            task.wait(note.delay)
        end
        playing = false
        N("Piano", "演奏終了", 3)
    end)
end

local function StopSong()
    playing = false
    N("Piano", "停止", 2)
end

-- ==============================
-- Orion GUI
-- ==============================
local SaSaHub = loadstring(game:HttpGet("https://raw.githubusercontent.com/jadpy/suki/refs/heads/main/orion"))()
if not SaSaHub then error("Orion Load Failed") end

local W = SaSaHub:MakeWindow({
    Name = "🎹 Piano",
    HidePremium = false,
    SaveConfig = false,
    IntroEnabled = false
})

local T = W:MakeTab({Name = "Piano", Icon = "rbxassetid://4483345998"})

T:AddSection({Name = "固定"})

T:AddToggle({
    Name = "体に固定",
    Default = false,
    Color = Color3.fromRGB(120, 255, 150),
    Callback = function(v)
        if v then StartFollow() else StopFollow() end
    end
})

T:AddSection({Name = "曲選択"})

T:AddDropdown({
    Name = "曲",
    Default = "MONTAGEM HIKARI",
    Options = {"MONTAGEM HIKARI", "星降る海", "NO_BATIDÃO", "YAJU&U", "のだ", "旅立ちの日に", "Bling Bang Bang Born", "Undertale", "怪獣 (サカナクション)"},
    Callback = function(v) currentSong = v end
})

T:AddSection({Name = "再生"})

T:AddButton({Name = "▶ 演奏開始", Callback = function() PlaySong() end})
T:AddButton({Name = "⏹ 停止", Callback = function() StopSong() end})

T:AddSection({Name = "テスト"})

T:AddButton({Name = "🎹 Cキー鳴らす", Callback = function()
    local kb = GetKeyboard()
    if kb then PressKey(kb, "Key1C"); N("Piano", "C鳴らした", 2) end
end})

SaSaHub:Init()
SaSaHub:MakeNotification({Name = "Piano", Content = "Ready. 曲 9 曲", Time = 3})
